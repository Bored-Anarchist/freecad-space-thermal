cmake_minimum_required(VERSION 3.22)
project(FCST VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
option(FCST_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
set(FCST_SOURCE_REVISION "uncommitted" CACHE STRING "Sanitized source commit hash")
if(NOT FCST_SOURCE_REVISION MATCHES "^(uncommitted|[0-9a-f]+)$")
  message(FATAL_ERROR "FCST_SOURCE_REVISION must be a commit hash")
endif()
configure_file(engine/include/fcst/build_info.hpp.in generated/fcst/build_info.hpp @ONLY)
add_library(fcst_core engine/src/diagnostic.cpp engine/src/contracts.cpp engine/src/thermal.cpp engine/src/thermal_assembly.cpp engine/src/thermal_steady.cpp engine/src/thermal_transient.cpp engine/src/radiation.cpp)
target_link_libraries(fcst_core PRIVATE bcrypt)
target_compile_features(fcst_core PUBLIC cxx_std_20)
set_target_properties(fcst_core PROPERTIES CXX_EXTENSIONS OFF)
target_include_directories(fcst_core PUBLIC engine/include "${CMAKE_CURRENT_BINARY_DIR}/generated")
add_executable(fcst-worker engine/worker/main.cpp)
target_link_libraries(fcst-worker PRIVATE fcst_core)
set(FCST_TARGETS fcst_core fcst-worker)
if(BUILD_TESTING)
  add_executable(fcst-thermal-tests engine/tests/thermal_tests.cpp)
  target_link_libraries(fcst-thermal-tests PRIVATE fcst_core)
  add_executable(fcst-thermal-assembly-tests engine/tests/thermal_assembly_tests.cpp)
  target_link_libraries(fcst-thermal-assembly-tests PRIVATE fcst_core)
  add_executable(fcst-thermal-steady-tests engine/tests/thermal_steady_tests.cpp)
  target_link_libraries(fcst-thermal-steady-tests PRIVATE fcst_core)
  add_executable(fcst-thermal-transient-tests engine/tests/thermal_transient_tests.cpp)
  target_link_libraries(fcst-thermal-transient-tests PRIVATE fcst_core)
  add_executable(fcst-thermal-reference-subject engine/tests/thermal_reference_subject.cpp)
  target_link_libraries(fcst-thermal-reference-subject PRIVATE fcst_core)
  add_executable(fcst-radiation-tests engine/tests/radiation_tests.cpp)
  target_link_libraries(fcst-radiation-tests PRIVATE fcst_core)
  add_executable(fcst-radiation-subject engine/tests/radiation_subject.cpp)
  target_link_libraries(fcst-radiation-subject PRIVATE fcst_core psapi)
  list(APPEND FCST_TARGETS fcst-radiation-tests fcst-radiation-subject)
  list(APPEND FCST_TARGETS fcst-thermal-tests fcst-thermal-assembly-tests fcst-thermal-steady-tests fcst-thermal-transient-tests fcst-thermal-reference-subject)
endif()
foreach(target ${FCST_TARGETS})
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive- /fp:strict /experimental:deterministic
      "/pathmap:${CMAKE_SOURCE_DIR}=src" "/pathmap:${CMAKE_BINARY_DIR}=build")
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror
      -fno-fast-math -ffp-contract=off
      "-ffile-prefix-map=${CMAKE_SOURCE_DIR}=src" "-ffile-prefix-map=${CMAKE_BINARY_DIR}=build")
    if(FCST_SANITIZERS)
      target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
  endif()
endforeach()
install(TARGETS fcst-worker RUNTIME DESTINATION bin)
install(FILES ci/dependencies.json DESTINATION share/fcst)
if(BUILD_TESTING)
  foreach(case intersections bvh-equivalence geometry-mapping seed-streams occlusion-groups exchange-correction black-gray-sphere jacobian thermal-coupling limits-failures sparse-storage)
    add_test(NAME radiation-${case} COMMAND fcst-radiation-tests ${case} "${CMAKE_SOURCE_DIR}/tests/fixtures/thermal")
  endforeach()
  add_test(NAME native-build-info COMMAND fcst-worker --build-info)
  set_tests_properties(native-build-info PROPERTIES PASS_REGULAR_EXPRESSION "fcst_native")
  add_test(NAME native-self-test COMMAND fcst-worker --self-test)
  foreach(case constant-law linear-law previous-law clamped-law invalid-laws storage tensor immutable-loading case-loading)
    add_test(NAME thermal-${case} COMMAND fcst-thermal-tests ${case} "${CMAKE_SOURCE_DIR}/tests/fixtures/thermal/material-network.json")
  endforeach()
  foreach(case balance algebraic-boundary enthalpy-step jacobian-differences callback-contracts boundary-callback singular-components sparsity backend-linear backend-nonlinear assembly-safety physics-coverage)
    add_test(NAME thermal-assembly-${case} COMMAND fcst-thermal-assembly-tests ${case} "${CMAKE_SOURCE_DIR}/tests/fixtures/thermal")
  endforeach()
  foreach(case lu-analytic lu-sparse lu-subnormal-tail linear nonlinear line-search domain-backtracking scaling singular limits energy stopping options)
    add_test(NAME thermal-steady-${case} COMMAND fcst-thermal-steady-tests ${case} "${CMAKE_SOURCE_DIR}/tests/fixtures/thermal")
  endforeach()
  foreach(case analytic two-node stiff initialization enthalpy independent-reference refinement rejection sampling heater event-refinement network-profile network-localized schedule simultaneous checkpoint chattering energy results profiling limits failures)
    add_test(NAME thermal-transient-${case} COMMAND fcst-thermal-transient-tests ${case} "${CMAKE_SOURCE_DIR}/tests/fixtures/thermal")
  endforeach()
endif()
