cmake_minimum_required(VERSION 3.22)
project(FCST VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
option(FCST_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
set(FCST_SOURCE_REVISION "uncommitted" CACHE STRING "Sanitized source commit hash")
if(NOT FCST_SOURCE_REVISION MATCHES "^(uncommitted|[0-9a-f]+)$")
  message(FATAL_ERROR "FCST_SOURCE_REVISION must be a commit hash")
endif()
configure_file(engine/include/fcst/build_info.hpp.in generated/fcst/build_info.hpp @ONLY)
add_library(fcst_core engine/src/diagnostic.cpp)
target_compile_features(fcst_core PUBLIC cxx_std_20)
set_target_properties(fcst_core PROPERTIES CXX_EXTENSIONS OFF)
target_include_directories(fcst_core PUBLIC engine/include "${CMAKE_CURRENT_BINARY_DIR}/generated")
add_executable(fcst-worker engine/worker/main.cpp)
target_link_libraries(fcst-worker PRIVATE fcst_core)
foreach(target fcst_core fcst-worker)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive- /fp:strict
      "/pathmap:${CMAKE_SOURCE_DIR}=src" "/pathmap:${CMAKE_BINARY_DIR}=build")
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror
      -fno-fast-math -ffp-contract=off
      "-ffile-prefix-map=${CMAKE_SOURCE_DIR}=src" "-ffile-prefix-map=${CMAKE_BINARY_DIR}=build")
    if(FCST_SANITIZERS)
      target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
  endif()
endforeach()
install(TARGETS fcst-worker RUNTIME DESTINATION bin)
install(FILES ci/dependencies.json DESTINATION share/fcst)
if(BUILD_TESTING)
  add_test(NAME native-build-info COMMAND fcst-worker --build-info)
  set_tests_properties(native-build-info PROPERTIES PASS_REGULAR_EXPRESSION "fcst_native")
  add_test(NAME native-self-test COMMAND fcst-worker --self-test)
endif()
