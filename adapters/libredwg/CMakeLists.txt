cmake_minimum_required(VERSION 3.22)
project(fcst_independent_dwg LANGUAGES C CXX)
if(NOT WIN32)
  message(FATAL_ERROR "The independent DWG build currently targets Windows")
endif()
foreach(required DWG_SOURCE FCST_PATCH_SHA256)
  if(NOT DEFINED ${required})
    message(FATAL_ERROR "Missing pinned backend build input")
  endif()
endforeach()
file(TO_CMAKE_PATH "${DWG_SOURCE}" DWG_SOURCE)
if(NOT CMAKE_C_COMPILER_ID STREQUAL "GNU" OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This frozen backend requires the pinned native Windows x64 GCC profile")
endif()
configure_file(config-windows.h "${CMAKE_CURRENT_BINARY_DIR}/config/config.h" COPYONLY)
set(backend_sources dwg common codepages bits logging decode decode2 decode_r11
    decode_r2007 reedsolomon print free hash dynapi classes dwg_api objects geom
    out_dxf out_dxfb out_json out_geojson encode encode2 dxfclasses in_dxf in_json)
list(TRANSFORM backend_sources PREPEND "${DWG_SOURCE}/src/")
list(TRANSFORM backend_sources APPEND ".c")
add_library(fcst_libredwg STATIC ${backend_sources})
target_include_directories(fcst_libredwg PUBLIC "${DWG_SOURCE}/include" "${DWG_SOURCE}/src" "${CMAKE_CURRENT_BINARY_DIR}/config")
target_compile_definitions(fcst_libredwg PRIVATE HAVE_CONFIG_H)
target_compile_options(fcst_libredwg PRIVATE -O1 -g0 -fno-fast-math -ffp-contract=off)
add_executable(fcst-dwg main.cpp)
target_compile_features(fcst-dwg PRIVATE cxx_std_20)
target_compile_definitions(fcst-dwg PRIVATE FCST_PATCH_SHA256="${FCST_PATCH_SHA256}")
target_compile_options(fcst-dwg PRIVATE -O2 -g0 -fno-fast-math -ffp-contract=off)
target_link_options(fcst-dwg PRIVATE -static -static-libgcc -static-libstdc++)
target_link_libraries(fcst-dwg PRIVATE fcst_libredwg m)
install(TARGETS fcst-dwg RUNTIME DESTINATION bin)
install(FILES LICENSE README.md DESTINATION share/fcst-dwg)
